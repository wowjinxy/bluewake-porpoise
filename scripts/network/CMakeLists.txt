cmake_minimum_required(VERSION 3.20)
project(BlueWakeNetworkStandalone LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(BW_NETWORK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
include("${BW_NETWORK_ROOT}/cmake/BlueWakeNetwork.cmake")
add_executable(bluewake_network_server server_main.cpp standalone_memory.c)
target_link_libraries(bluewake_network_server PRIVATE bluewake_network)
add_executable(bluewake_network_core_test "${BW_NETWORK_ROOT}/tests/network_session_test.cpp" standalone_memory.c)
target_link_libraries(bluewake_network_core_test PRIVATE bluewake_network)
add_executable(bluewake_progression_sync_test "${BW_NETWORK_ROOT}/tests/progression_sync_test.c" standalone_memory.c)
target_link_libraries(bluewake_progression_sync_test PRIVATE bluewake_network)
add_executable(bluewake_network_game_test "${BW_NETWORK_ROOT}/tests/network_game_test.cpp" standalone_memory.c)
target_link_libraries(bluewake_network_game_test PRIVATE bluewake_network)
add_executable(bluewake_network_game_backpressure_test "${BW_NETWORK_ROOT}/tests/network_game_backpressure_test.cpp" standalone_memory.c)
target_link_libraries(bluewake_network_game_backpressure_test PRIVATE bluewake_network)
enable_testing()
add_test(NAME bluewake_progression_sync_test COMMAND bluewake_progression_sync_test)
add_test(NAME bluewake_network_core_test COMMAND bluewake_network_core_test)
add_test(NAME bluewake_network_game_test COMMAND bluewake_network_game_test)
add_test(NAME bluewake_network_personal_test COMMAND bluewake_network_game_test --personal)
add_test(NAME bluewake_network_game_backpressure_test COMMAND bluewake_network_game_backpressure_test)
