cmake_minimum_required(VERSION 3.24)
project(bluewake_cache_callback_oracle C)
enable_testing()
option(BLUEWAKE_TEST_ASAN "Instrument the authored fixture and all CPU TUs" OFF)
set(BLUEWAKE_REPO "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "BlueWake repository")
get_filename_component(BLUEWAKE_REPO "${BLUEWAKE_REPO}" ABSOLUTE)
set(CORE "${BLUEWAKE_REPO}/ref/recompcore/GXRuntime")
set(SOURCES cpu.c cpu_exception.c cpu_interpreter.c cpu_interpreter_float.c cpu_interpreter_integer.c cpu_interpreter_table.c)
list(TRANSFORM SOURCES PREPEND "${CORE}/src/core/")
add_executable(cache_callback_test ${SOURCES}
  "${BLUEWAKE_REPO}/cmake/composite/gather_pipe.c" cache_callback_test.c)
set_source_files_properties("${CORE}/src/core/cpu.c" PROPERTIES
  COMPILE_DEFINITIONS ppc_fallback_instruction=bw_reference_runtime_fallback)
target_include_directories(cache_callback_test PRIVATE "${CORE}/include" "${BLUEWAKE_REPO}/cmake/composite")
target_compile_features(cache_callback_test PRIVATE c_std_11)
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  message(FATAL_ERROR "This authored oracle requires Clang or GCC")
endif()
target_compile_options(cache_callback_test PRIVATE -O3 -UNDEBUG -ffp-contract=off)
if(WIN32 AND CMAKE_C_COMPILER_ID STREQUAL "Clang")
  target_compile_options(cache_callback_test PRIVATE -fms-compatibility-version=19.44 -fms-runtime-lib=dll)
  target_link_options(cache_callback_test PRIVATE -fms-runtime-lib=dll)
elseif(NOT WIN32)
  target_link_libraries(cache_callback_test PRIVATE m)
endif()
if(BLUEWAKE_TEST_ASAN)
  target_compile_options(cache_callback_test PRIVATE -fsanitize=address -fno-omit-frame-pointer)
  target_link_options(cache_callback_test PRIVATE -fsanitize=address)
endif()
add_test(NAME cache_callback_test COMMAND cache_callback_test)
